# CEMAPI — crontab -e ile ekle. Saatler İstanbul (Europe/Istanbul).
# A2#05 / V2: :01 eski slot kapanır, :02 yeni işlem açılır.

CRON_TZ=Europe/Istanbul
COPTC=/root/projects/CEMAPI

# :01  kaynak slot kapandı — PM settle + redeem
1  * * * * cd $COPTC && ./venv/bin/python runner.py close     >> /tmp/coptc.log 2>&1

# :02  yeni kaynak işlemi — :02:00…:09:00 API poll (4 sn) + PM emri
2  * * * * cd $COPTC && ./venv/bin/python runner.py live-open >> /tmp/coptc.log 2>&1

# :12  PM sonucu geciktiğinde kapanış tekrarı
12 * * * * cd $COPTC && ./venv/bin/python runner.py settle    >> /tmp/coptc.log 2>&1
